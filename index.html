(() => { const splash = document.querySelector(".splash-screen"); if (!splash) return; // Duración de la animación. const tiempoEntrada = 1200; const tiempoVisible = 1800; const tiempoSalida = 800; // Comprobar si se está realizando una recuperación de contraseña. const parametros = new URLSearchParams(window.location.search); const hash = new URLSearchParams( window.location.hash.replace(/^#/, "") ); const esRecuperacion = parametros.get("type") === "recovery" || hash.get("type") === "recovery" || hash.has("access_token") || hash.has("error_code"); // Esperar a que termine la animación. window.setTimeout(() => { splash.classList.add("is-leaving"); window.setTimeout(() => { if (esRecuperacion) { window.location.replace( "../01-login/recuperar.html" + window.location.search + window.location.hash ); } else { window.location.replace("../01-login/index.html"); } }, tiempoSalida); }, tiempoEntrada + tiempoVisible); })();
